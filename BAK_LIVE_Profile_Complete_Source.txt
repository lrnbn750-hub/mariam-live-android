    private fun loadProfileData() {
        val token = getSharedPreferences("MariamLive", MODE_PRIVATE).getString("token", null)
        if (token.isNullOrBlank()) { finish(); return }
        executor.execute {
            try {
                val me = AuthApi.get("/api/auth/me", token)
                if (me.status !in 200..299) throw Exception(me.body.optString("error", "تعذر تحميل الحساب"))
                val account = me.body.optJSONObject("user") ?: throw Exception("بيانات الحساب غير مكتملة")
                userId = account.optString("id")
                publicId = account.optString("public_id")
                if (userId.isBlank() || userId == "null") throw Exception("معرّف المستخدم غير متوفر")
                val details = AuthApi.get("/api/users/$userId", token)
                if (details.status !in 200..299) throw Exception(details.body.optString("error", "تعذر تحميل الملف الشخصي"))
                val user = details.body.optJSONObject("user") ?: JSONObject()
                val data = JSONObject()
                    .put("name", user.optString("display_name").takeIf { it.isNotBlank() && it != "null" } ?: user.optString("username", "مستخدم BAK LIVE"))
                    .put("id", publicId)
                    .put("level", 1)
                    .put("avatarUrl", user.optString("avatar_url"))
                    .put("visitors", JSONObject.NULL)
                    .put("friends", 0)
                    .put("followers", 0)
                    .put("fans", 0)
                    .put("diamonds", 0)
                    .put("coins", 0)
                try {
                    val level = AuthApi.get("/api/levels/me", token)
                    if (level.status in 200..299) data.put("level", level.body.optInt("level", 1).coerceIn(1, 100))
                } catch (_: Exception) {}
                try {
                    val follows = AuthApi.get("/api/follows/me", token)
                    if (follows.status in 200..299) {
                        data.put("followers", follows.body.optInt("following_count", follows.body.optJSONArray("following")?.length() ?: 0))
                        data.put("fans", follows.body.optInt("followers_count", follows.body.optJSONArray("followers")?.length() ?: 0))
                    }
                } catch (_: Exception) {}
                try {
                    val wallet = AuthApi.get("/api/wallet", token)
                    if (wallet.status in 200..299) {
                        val w = wallet.body.optJSONObject("wallet") ?: wallet.body
                        data.put("coins", w.optLong("recharge_coins", 0L))
                        data.put("diamonds", w.optLong("earned_diamonds", 0L))
                    }
                } catch (_: Exception) {}
                profileData = data
                runOnUiThread { bindProfile(data, token) }
            } catch (e: Exception) {
                Log.e("BAKProfileNative", "Profile load failed", e)
                runOnUiThread { Toast.makeText(this, e.message ?: "تعذر تحميل بيانات الملف الشخصي", Toast.LENGTH_LONG).show() }
            }
        }
    }

    private fun bindProfile(data: JSONObject, token: String) {
        header.setUserName(data.optString("name"))
        header.setUserIdText(publicId)
        header.setLevel(data.optInt("level", 1))
        header.setFansCount(data.optInt("fans", 0).toString())
        header.setFollowingCount(data.optInt("followers", 0).toString())
        coinValue.text = data.optLong("coins", 0).toString()
        diamondValue.text = data.optLong("diamonds", 0).toString()
        val avatarUrl = normalizeAvatar(data.optString("avatarUrl"))
        if (avatarUrl.isNotBlank()) loadAvatar(avatarUrl, token)
    }

    private fun normalizeAvatar(value: String): String {
        if (value.isBlank() || value == "null") return ""
        return if (value.startsWith("http://") || value.startsWith("https://")) value
        else "https://nokiraq.duckdns.org/" + value.trimStart('/')
    }

    private fun loadAvatar(url: String, token: String) {
        executor.execute {
            try {
                val connection = (URL(url).openConnection() as java.net.HttpURLConnection).apply {
                    connectTimeout = 15000; readTimeout = 20000; instanceFollowRedirects = true
                    setRequestProperty("Accept", "image/*")
                    if (token.isNotBlank()) setRequestProperty("Authorization", "Bearer $token")
                    setRequestProperty("User-Agent", "BAK-LIVE-Android")
                }
                val bitmap = try {
                    val status = connection.responseCode
                    if (status !in 200..299) throw Exception("Avatar HTTP $status")
                    val mime = connection.contentType?.substringBefore(';')?.trim().orEmpty()
                    if (!mime.startsWith("image/")) throw Exception("Avatar response is not image: $mime")
                    val bytes = connection.inputStream.use { it.readBytes() }
                    BitmapFactory.decodeByteArray(bytes, 0, bytes.size) ?: throw Exception("Unsupported avatar image")
                } finally { connection.disconnect() }
                runOnUiThread { if (!isFinishing) avatar.setImageBitmap(bitmap) }
            } catch (e: Exception) {
                Log.e("BAKProfileNative", "Avatar load failed: $url", e)
            }
        }
    }

    private fun copyId() {

    }

    private fun openFollowList(type: String) {
        val token = getSharedPreferences("MariamLive", MODE_PRIVATE).getString("token", null) ?: return
        val title = if (type == "followers") "المعجبون" else "أتابعهم"
        val dialog = android.app.AlertDialog.Builder(this).setTitle(title)
            .setMessage("جارٍ تحميل القائمة...").setNegativeButton("إغلاق", null).create()
        dialog.show()
        executor.execute {
            try {
                val result = AuthApi.get("/api/follows/me", token)
                if (result.status !in 200..299) throw Exception(result.body.optString("error", "تعذر تحميل القائمة"))
                val list = result.body.optJSONArray(type) ?: org.json.JSONArray()
                val names = ArrayList<String>()
                for (i in 0 until list.length()) {
                    val item = list.optJSONObject(i) ?: continue
                    names.add(item.optString("display_name").takeIf { it.isNotBlank() && it != "null" } ?: item.optString("username", "مستخدم"))
                }
                runOnUiThread {
                    if (isFinishing) return@runOnUiThread
                    dialog.dismiss()
                    android.app.AlertDialog.Builder(this).setTitle("$title (${names.size})")
                        .setMessage(if (names.isEmpty()) "لا توجد حسابات في هذه القائمة بعد" else names.joinToString("\n"))
                        .setPositiveButton("حسناً", null).show()
                }
            } catch (e: Exception) {
                runOnUiThread { dialog.dismiss(); Toast.makeText(this, e.message ?: "تعذر تحميل القائمة", Toast.LENGTH_SHORT).show() }
            }
        }
    }

    private fun pickPhoto() {
        val intent = Intent(Intent.ACTION_OPEN_DOCUMENT).apply {
            addCategory(Intent.CATEGORY_OPENABLE)
            type = "image/*"
        }
        startActivityForResult(intent, photoPickerCode)
    }

        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == photoPickerCode && resultCode == RESULT_OK) {
            val uri = data?.data ?: return
            uploadPhoto(uri)
        }
    }

    private fun uploadPhoto(uri: Uri) {
        val token = getSharedPreferences("MariamLive", MODE_PRIVATE).getString("token", null) ?: return
        Toast.makeText(this, "جارٍ رفع الصورة...", Toast.LENGTH_SHORT).show()
        executor.execute {
            try {
                val source = contentResolver.openInputStream(uri)?.use { it.readBytes() } ?: throw Exception("تعذر قراءة الصورة")
                val bounds = BitmapFactory.Options().apply { inJustDecodeBounds = true }
                BitmapFactory.decodeByteArray(source, 0, source.size, bounds)
                var sample = 1
                while (bounds.outWidth / sample > 1200 || bounds.outHeight / sample > 1200) sample *= 2
                val bitmap = BitmapFactory.decodeByteArray(source, 0, source.size,
                    BitmapFactory.Options().apply { inSampleSize = sample }) ?: throw Exception("صيغة الصورة غير مدعومة")
                val scaled = if (bitmap.width > 900 || bitmap.height > 900) {
                    val ratio = minOf(900f / bitmap.width, 900f / bitmap.height)
                    Bitmap.createScaledBitmap(bitmap, (bitmap.width * ratio).toInt().coerceAtLeast(1),
                        (bitmap.height * ratio).toInt().coerceAtLeast(1), true)
                } else bitmap
                val output = ByteArrayOutputStream()
                scaled.compress(Bitmap.CompressFormat.JPEG, 82, output)
                val bytes = output.toByteArray()
                if (bytes.size > 1536 * 1024) throw Exception("الصورة كبيرة، اختر صورة أصغر")
                val result = AuthApi.post("/api/profile/avatar",
                    JSONObject().put("avatar_base64", Base64.encodeToString(bytes, Base64.NO_WRAP)), token)
                if (result.status !in 200..299) throw Exception(result.body.optString("error", "تعذر رفع الصورة"))
                val url = normalizeAvatar(result.body.optString("avatar_url"))
                profileData.put("avatarUrl", url)
                runOnUiThread {
                    loadAvatar(url, token)
                    Toast.makeText(this, "تم تحديث الصورة الشخصية", Toast.LENGTH_SHORT).show()
                }
                if (scaled !== bitmap) scaled.recycle()
                if (bitmap !== scaled) bitmap.recycle()
            } catch (e: Exception) {
                runOnUiThread { Toast.makeText(this, e.message ?: "تعذر رفع الصورة", Toast.LENGTH_LONG).show() }
            }
        }
    }

    override fun onDestroy() {
        executor.shutdownNow()
        super.onDestroy()
    }

/**
 * BAK LIVE level badge, levels 1..100. Pure Canvas drawing: no images, no libraries, crisp at any density.
 *
 * 10 tiers x 10 levels. Each tier has its own palette + emblem
 * (Steel, Emerald, Aqua, Sapphire, Amethyst, Rose, Flame, Gold, Platinum, Royal).
 * Inside a tier the badge "levels up" visually: sub-level 4+ gets a thicker rim, 6+ a soft glow,
 * 9 a glitter spark; the last tier (91-100) also gets a sun-burst emblem, sparkles and a shimmer sweep.
 *
 * UI only: set [level] from the value you already load from your API.
 *
 *   levelBadge.level = user.level          // 1..100, clamped
 */
class LevelBadgeView @JvmOverloads constructor(
    context: Context,
    attrs: AttributeSet? = null,
    defStyleAttr: Int = 0
) : View(context, attrs, defStyleAttr) {

    private class Tier(
        val pillA: Int, val pillB: Int, val stroke: Int,
        val ringA: Int, val ringB: Int, val innerA: Int, val innerB: Int,
        val glyph: Int, val text: Int, val glow: Int, val shape: Int, val textShadow: Int
    )

    private val d = resources.displayMetrics.density
    private val paint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val labelPaint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val numPaint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val path = Path()
    private val glyphPath = Path()
    private val rect = RectF()

    private var levelValue = 1
    private var tierIdx = 0
    private var sub = 0
    private var labelW = 0f
    private var numW = 0f
    private var shimmerPhase = 0.55f
    private var animator: ValueAnimator? = null

    // geometry (dp)
    private val emblemR = 11f * d
    private val pillH = 16f * d
    private val extraX = 4f * d
    private val extraY = 3f * d
    private val textGap = 3f * d
    private val padEnd = 8f * d
    private val labelToNum = 2f * d

    /** Sweeping shine for levels 81..100. Set false to keep the badge fully static. */
    var shimmerEnabled: Boolean = true
        set(value) { field = value; updateAnimator(); invalidate() }

    /** 1..100 (values outside are clamped). */
    var level: Int
        get() = levelValue
        set(value) {
            levelValue = value.coerceIn(1, 100)
            tierIdx = (levelValue - 1) / 10
            sub = (levelValue - 1) % 10
            contentDescription = "Level $levelValue"
            measureTexts()
            requestLayout()
            updateAnimator()
            invalidate()
        }

    init {
        labelPaint.typeface = Typeface.create(Typeface.DEFAULT, Typeface.BOLD_ITALIC)
        labelPaint.textSize = 8f * d
        numPaint.typeface = Typeface.create(Typeface.DEFAULT, Typeface.BOLD_ITALIC)
        numPaint.textSize = 12f * d
        level = 1
    }

    private fun measureTexts() {
        labelW = labelPaint.measureText("LV")
        numW = numPaint.measureText(levelValue.toString())
    }

    override fun onMeasure(widthMeasureSpec: Int, heightMeasureSpec: Int) {
        val w = extraX + emblemR * 2f + textGap + labelW + labelToNum + numW + padEnd + extraX
        val h = emblemR * 2f + extraY * 2f
        setMeasuredDimension(
            resolveSize(ceil(w).toInt(), widthMeasureSpec),
            resolveSize(ceil(h).toInt(), heightMeasureSpec)
        )
    }

    // ------------------------------------------------------------------ animation
    private fun updateAnimator() {
        val shouldRun = shimmerEnabled && levelValue >= 81 && isAttachedToWindow && visibility == VISIBLE
        if (shouldRun) {
            if (animator == null) {
                animator = ValueAnimator.ofFloat(0f, 1.7f).apply {
                    duration = 2800L
                    repeatCount = ValueAnimator.INFINITE
                    interpolator = LinearInterpolator()
                    addUpdateListener {
                        shimmerPhase = (it.animatedValue as Float).coerceAtMost(1f)
                        invalidate()
                    }
                    start()
                }
            }
        } else {
            animator?.cancel()
            animator = null
        }
    }

    override fun onAttachedToWindow() { super.onAttachedToWindow(); updateAnimator() }
    override fun onDetachedFromWindow() { animator?.cancel(); animator = null; super.onDetachedFromWindow() }
    override fun onVisibilityChanged(changedView: View, visibility: Int) {
        super.onVisibilityChanged(changedView, visibility); updateAnimator()
    }

    // ------------------------------------------------------------------ drawing
    private fun withAlpha(color: Int, a: Int): Int = (color and 0x00FFFFFF) or (a.coerceIn(0, 255) shl 24)

    private fun buildGlyph(ops: FloatArray, cx: Float, cy: Float, s: Float, out: Path) {
        out.reset()
        var i = 0
        while (i < ops.size) {
            when (ops[i].toInt()) {
                0 -> { out.moveTo(cx + ops[i + 1] * s, cy + ops[i + 2] * s); i += 3 }
                1 -> { out.lineTo(cx + ops[i + 1] * s, cy + ops[i + 2] * s); i += 3 }
                2 -> {
                    out.quadTo(cx + ops[i + 1] * s, cy + ops[i + 2] * s, cx + ops[i + 3] * s, cy + ops[i + 4] * s)
                    i += 5
                }
                else -> { out.close(); i += 1 }
            }
        }
    }

    private fun drawSparkle(c: Canvas, x: Float, y: Float, r: Float, color: Int) {
        buildGlyph(G_SPARKLE, x, y, r, glyphPath)
        paint.shader = null
        paint.style = Paint.Style.FILL
        paint.color = color
        c.drawPath(glyphPath, paint)
    }

    override fun onDraw(canvas: Canvas) {
        val t = TIERS[tierIdx]
        val cx = extraX + emblemR
        val cy = height / 2f
        val pillL = cx
        val pillR = width - extraX
        val pillT = cy - pillH / 2f
        val pillB = cy + pillH / 2f
        val rad = pillH / 2f
        val isRoyal = tierIdx == 9

        // 1) soft glow (no blur filters -> works with hardware acceleration on every API)
        val glowLayers = when {
            sub >= 8 -> 3
            sub >= 5 -> 2
            isRoyal -> 2
            else -> 0
        }
        paint.shader = null
        paint.style = Paint.Style.STROKE
        for (k in glowLayers downTo 1) {
            paint.strokeWidth = k * 1.7f * d
            paint.color = withAlpha(t.glow, 95 / k)
            val e = k * 0.85f * d
            rect.set(pillL - e, pillT - e, pillR + e, pillB + e)
            canvas.drawRoundRect(rect, rad + e, rad + e, paint)
        }

        // 2) sun-burst rays behind the emblem (tier 10 only)
        if (t.shape == 6) {
            paint.style = Paint.Style.STROKE
            paint.strokeWidth = 1.3f * d
            paint.strokeCap = Paint.Cap.ROUND
            paint.color = withAlpha(t.glow, 210)
            for (i in 0 until 12) {
                val a = Math.toRadians(i * 30.0)
                val r1 = emblemR + 0.6f * d
                val r2 = emblemR + (if (i % 2 == 0) 3.0f else 2.0f) * d
                canvas.drawLine(
                    cx + (cos(a) * r1).toFloat(), cy + (sin(a) * r1).toFloat(),
                    cx + (cos(a) * r2).toFloat(), cy + (sin(a) * r2).toFloat(), paint
                )
            }
            paint.strokeCap = Paint.Cap.BUTT
        }

        // 3) pill body
        rect.set(pillL, pillT, pillR, pillB)
        paint.style = Paint.Style.FILL
        paint.shader = LinearGradient(pillL, 0f, pillR, 0f, t.pillA, t.pillB, Shader.TileMode.CLAMP)
        canvas.drawRoundRect(rect, rad, rad, paint)
        // depth: darker bottom edge
        paint.shader = LinearGradient(0f, pillT, 0f, pillB, 0x00000000, 0x30000000, Shader.TileMode.CLAMP)
        canvas.drawRoundRect(rect, rad, rad, paint)

        // gloss + shimmer, clipped to the pill
        canvas.save()
        path.reset()
        path.addRoundRect(rect, rad, rad, Path.Direction.CW)
        canvas.clipPath(path)
        paint.shader = LinearGradient(0f, pillT, 0f, cy, 0x66FFFFFF, 0x00FFFFFF, Shader.TileMode.CLAMP)
        canvas.drawRect(pillL, pillT, pillR, cy, paint)
        if (shimmerEnabled && levelValue >= 81) {
            val bw = 14f * d
            val sx = pillL - bw + (pillR - pillL + 2f * bw) * shimmerPhase
            paint.shader = LinearGradient(
                sx - bw, pillT, sx + bw, pillB,
                intArrayOf(0x00FFFFFF, 0x77FFFFFF, 0x00FFFFFF), floatArrayOf(0f, 0.5f, 1f), Shader.TileMode.CLAMP
            )
            canvas.drawRect(pillL, pillT, pillR, pillB, paint)
        }
        canvas.restore()

        // rim
        paint.shader = null
        paint.style = Paint.Style.STROKE
        paint.strokeWidth = (if (sub >= 3) 1.5f else 1.0f) * d
        paint.color = t.stroke
        rect.set(pillL + paint.strokeWidth / 2f, pillT + paint.strokeWidth / 2f, pillR - paint.strokeWidth / 2f, pillB - paint.strokeWidth / 2f)
        canvas.drawRoundRect(rect, rad, rad, paint)

        // 4) text "LV" + number
        val baseY = cy - (numPaint.fontMetrics.ascent + numPaint.fontMetrics.descent) / 2f
        val tx = cx + emblemR + textGap
        val nx = tx + labelW + labelToNum
        val sdy = 0.6f * d
        labelPaint.color = t.textShadow; canvas.drawText("LV", tx, baseY + sdy, labelPaint)
        numPaint.shader = null
        numPaint.color = t.textShadow; canvas.drawText(levelValue.toString(), nx, baseY + sdy, numPaint)
        labelPaint.color = withAlpha(t.text, 215); canvas.drawText("LV", tx, baseY, labelPaint)
        if (isRoyal) {
            numPaint.shader = LinearGradient(0f, cy - 6f * d, 0f, cy + 6f * d, 0xFFFFF3C0.toInt(), 0xFFFFC94D.toInt(), Shader.TileMode.CLAMP)
        } else {
            numPaint.color = t.text
        }
        canvas.drawText(levelValue.toString(), nx, baseY, numPaint)
        numPaint.shader = null

        // 5) emblem: metallic ring + inner disc + glyph
        paint.style = Paint.Style.FILL
        paint.shader = LinearGradient(cx - emblemR, cy - emblemR, cx + emblemR, cy + emblemR, t.ringA, t.ringB, Shader.TileMode.CLAMP)
        canvas.drawCircle(cx, cy, emblemR, paint)
        val innerR = emblemR - 2f * d
        paint.shader = LinearGradient(0f, cy - emblemR, 0f, cy + emblemR, t.innerA, t.innerB, Shader.TileMode.CLAMP)
        canvas.drawCircle(cx, cy, innerR, paint)
        canvas.save()
        path.reset()
        path.addCircle(cx, cy, innerR, Path.Direction.CW)
        canvas.clipPath(path)
        paint.shader = LinearGradient(0f, cy - emblemR, 0f, cy, 0x55FFFFFF, 0x00FFFFFF, Shader.TileMode.CLAMP)
        canvas.drawRect(cx - emblemR, cy - emblemR, cx + emblemR, cy, paint)
        canvas.restore()
        paint.shader = null
        paint.style = Paint.Style.STROKE
        paint.strokeWidth = 0.8f * d
        paint.color = 0x66FFFFFF
        canvas.drawCircle(cx, cy, emblemR - 0.4f * d, paint)

        val ops: FloatArray
        val gs: Float
        when (t.shape) {
            0 -> { ops = G_STAR; gs = 0.58f * emblemR }
            1, 5 -> { ops = G_GEM; gs = 0.50f * emblemR }
            2 -> { ops = G_SHIELD; gs = 0.50f * emblemR }
            3 -> { ops = G_BOLT; gs = 0.52f * emblemR }
            else -> { ops = G_CROWN; gs = 0.52f * emblemR }
        }
        buildGlyph(ops, cx, cy, gs, glyphPath)
        paint.style = Paint.Style.FILL
        paint.color = t.glyph
        canvas.drawPath(glyphPath, paint)
        if (t.shape == 1 || t.shape == 5) {
            buildGlyph(G_GEM_FACETS, cx, cy, gs, glyphPath)
            paint.style = Paint.Style.STROKE
            paint.strokeWidth = 0.7f * d
            paint.color = withAlpha(t.innerA, 170)
            canvas.drawPath(glyphPath, paint)
        }

        // 6) sparkles
        if (t.shape == 5) drawSparkle(canvas, cx + emblemR * 0.45f, cy - emblemR * 0.45f, emblemR * 0.30f, 0xFFFFFFFF.toInt())
        if (sub == 9 || isRoyal) {
            drawSparkle(canvas, pillR - 7f * d, pillT + 1.5f * d, 3f * d, if (isRoyal) 0xFFFFF0B0.toInt() else 0xEBFFFFFF.toInt())
        }
        if (isRoyal) {
            drawSparkle(canvas, cx + emblemR * 0.75f, cy - emblemR * 0.85f, 2.4f * d, 0xFFFFF0B0.toInt())
            if (levelValue == 100) drawSparkle(canvas, cx - emblemR * 0.7f, cy + emblemR * 0.85f, 2f * d, 0xFFFFF0B0.toInt())
        }
    }
